1 e 2- CRIAÇÃO DAS TABELAS:

SCHEMA: DESAFIO

CREATE TABLE regioes (
id_regioes INT(40) NOT NULL,
nome VARCHAR(40) NOT NULL,
estado VARCHAR(40) NOT NULL,
potencial_energetico VARCHAR(40) NOT NULL,
PRIMARY KEY(id_regioes)
);

CREATE TABLE usinas (
id_usinas INT(40) NOT NULL,
nome VARCHAR(40) NOT NULL,
tipo VARCHAR(40) NOT NULL,
capacidade VARCHAR(40) NOT NULL,
id_regioes INT(40) NOT NULL,
PRIMARY KEY(id_usinas),
FOREIGN KEY(id_regioes) REFERENCES regioes(id_regioes)
);

CREATE TABLE geracao (
id_geracao INT(40) NOT NULL,
id_usinas INT(40) NOT NULL,
data DATE NOT NULL,
energia_gerada DECIMAL(10,2) NOT NULL,
PRIMARY KEY(id_geracao),
FOREIGN KEY(id_usinas) REFERENCES usinas(id_usinas),
);

4- EXECUTAR CONSULTAS: 

UPDATE regioes 
SET estado = 'São Paulo'
WHERE id_regioes = 4;

SELECT * tipo 
FROM usinas
WHERE tipo IN ('Hidreletrica');


SELECT COUNT(energia_gerada) FROM geracao as contagem_energia

SELECT MAX(energia_gerada) FROM geracao as maximo_energia

SELECT MIN(energia_gerada) FROM geracao as minimo_energia

SELECT id_usinas, SUM(energia_gerada) as total_energia 
FROM geracao GROUP BY id_usinas HAVING SUM(energia_gerada) > 5000

SELECT  usinas.nome, geracao.energia_gerada
FROM usinas
INNER JOIN geracao
ON usinas.id_usinas = geracao.id_usinas
